cmake_minimum_required(VERSION 3.24)
project(meshtrain_consumer LANGUAGES CXX)

find_package(meshtrain 0.1 CONFIG REQUIRED)
if(DEFINED MESHTRAIN_EXPECTED_PREFIX)
    cmake_path(IS_PREFIX MESHTRAIN_EXPECTED_PREFIX "${meshtrain_DIR}" NORMALIZE correct_package)
    if(NOT correct_package)
        message(FATAL_ERROR "Consumer found a package outside the relocated install")
    endif()
endif()
add_executable(meshtrain_consumer main.cpp)
set_target_properties(meshtrain_consumer PROPERTIES CXX_EXTENSIONS OFF)
target_link_libraries(meshtrain_consumer PRIVATE meshtrain::core)
enable_testing()
add_test(NAME consume_installed_library COMMAND meshtrain_consumer)
